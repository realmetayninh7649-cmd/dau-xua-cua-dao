  function syncRoute(){

  async function loadKind(group,force=false){
    if(!force && cache.get(group.kind)?.ready)return;
    const ticket=++requestNumber;
    if(current?.kind===group.kind)byId('gdd-cat-status').textContent='Đang tải danh mục tư liệu…';
    const cfg=window.SITE_CONFIG||{};
    const endpoint=String(cfg.supabaseUrl||'').replace(/\/$/,'');
    const key=cfg.supabasePublishableKey;
    if(!endpoint||!key){
      if(current?.kind===group.kind)showError('Chưa kết nối Supabase. Vui lòng kiểm tra cấu hình website.');
      return;
    }
    const rows=[];
    try{
      for(let offset=0;offset<MAX_FETCH;offset+=FETCH_SIZE){
        const url=endpoint+'/rest/v1/'+encodeURIComponent(group.kind)+'?select='+encodeURIComponent(SELECTS[group.kind])+'&published=eq.true&order=created_at.desc&limit='+FETCH_SIZE+'&offset='+offset;
        const response=await fetch(url,{headers:{apikey:key,Accept:'application/json'},credentials:'omit'});
        if(!response.ok)throw Error('Lỗi Supabase '+response.status);
        const page=await response.json();
        if(!Array.isArray(page))throw Error('Kết quả máy chủ không hợp lệ');
        rows.push(...page);
        if(page.length<FETCH_SIZE)break;
      }
      cache.set(group.kind,{ready:true,rows});
      if(current?.kind===group.kind){
        refreshSubgroups();refreshTopics();renderRows();
      }
    }catch(err){
      if(current?.kind===group.kind)showError('Không thể tải danh mục ('+err.message+'). Hãy kiểm tra kết nối và thử lại.');
    }
  }

    byId('gdd-cat-pages').replaceChildren();
    byId('gdd-cat-retry')?.addEventListener('click',()=>loadKind(current,true));
  }
  function refreshSubgroups(){
    if(!cache.get('people')?.ready)return;
    const rows=cache.get('people')?.rows||[];
    const subset=rows.filter(p=>filters.branch==='all'||(p.branch||'Khác')===filters.branch);
    const values=[...new Set(subset.map(p=>p.subgroup).filter(Boolean))].sort((a,b)=>a.localeCompare(b,'vi'));
    const select=byId('gdd-cat-sub');
    select.innerHTML='<option value="all">Tất cả phân nhóm</option>'+values.map(v=>`<option value="${escape(v)}">${escape(v)}</option>`).join('');
    if(!values.includes(filters.subgroup))filters.subgroup='all';
    select.value=filters.subgroup;
  }
  function refreshTopics(){
    const rows=cache.get('posts')?.rows||[];
    const values=[...new Set(rows.map(p=>p.category).filter(Boolean))].sort((a,b)=>a.localeCompare(b,'vi'));
    byId('gdd-cat-topic').innerHTML='<option value="all">Tất cả chủ đề</option>'+values.map(v=>`<option value="${escape(v)}">${escape(v)}</option>`).join('');
    byId('gdd-cat-topic').value=values.includes(filters.category)?filters.category:'all';
  }
  function renderRows(){
    if(!current)return;
    const stored=cache.get(current.kind);
    if(!stored?.ready){
      byId('gdd-cat-grid').innerHTML='<div class="gdd-catalog-empty">Đang chuẩn bị kho tư liệu…</div>';
      byId('gdd-cat-pages').replaceChildren();
      return;
    }
    const words=norm(filters.query.trim());
    const rows=stored.rows.filter(p=>{
      if(current.kind==='people' && filters.branch!=='all' && (p.branch||'Khác')!==filters.branch)return false;
      if(current.kind==='people' && filters.subgroup!=='all' && p.subgroup!==filters.subgroup)return false;
      if(current.kind==='posts' && filters.category!=='all' && p.category!==filters.category)return false;
      if(!words)return true;
      const haystack=[p.full_name,p.secular_name,p.rank,p.branch,p.subgroup,p.years,p.birthplace,p.summary,
        p.title,p.category,p.body,p.caption,p.deceased_name,p.document_number,p.document_date,p.memorial_date,p.sources].filter(Boolean).join(' ');
      return norm(haystack).includes(words);
    });
    const pages=Math.max(1,Math.ceil(rows.length/PAGE_SIZE));
    filters.page=Math.min(Math.max(1,filters.page),pages);
    const start=(filters.page-1)*PAGE_SIZE;
    const shown=rows.slice(start,start+PAGE_SIZE);
    byId('gdd-cat-status').textContent=rows.length?`Hiển thị ${start+1}–${start+shown.length} trong ${rows.length} tư liệu · Trang ${filters.page}/${pages}`:'Không tìm thấy tư liệu phù hợp.';
    byId('gdd-cat-grid').innerHTML=shown.length?shown.map(p=>cardHTML(p,current.kind)).join(''):'<div class="gdd-catalog-empty">Chưa có tư liệu phù hợp. Thử đổi từ khóa hoặc bộ lọc.</div>';
    renderPages(pages);
  }
  function cardHTML(row,kind){
    const title=kind==='people'?row.full_name:row.title;

      kind==='obituaries'?row.memorial_date:kind==='documents'?[row.document_number,row.document_date].filter(Boolean).join(' · '):
      kind==='posts'?row.category:kind==='videos'?'VIDEO ĐẠO SỬ':'ẢNH TƯ LIỆU';
    const description=row.summary||row.caption||row.birthplace||'Tư liệu đang được biên soạn và bổ sung thông tin.';
    const docsImage=kind==='documents' && /\.(jpe?g|png|webp)(\?|$)/i.test(row.file_url||'')?row.file_url:'';
    const picture=safeURL(row.image_url||row.cover_url||docsImage);
    const poster=picture?`<img src="${escape(picture)}" alt="${escape(title)}" loading="lazy" decoding="async">`:
      `<span class="gdd-catalog-placeholder" aria-hidden="true"><span>✦</span>${escape(kind==='videos'?'VIDEO ĐẠO SỬ':kind==='documents'?'TÀI LIỆU ĐẠO SỬ':'DẤU XƯA CỬA ĐẠO')}</span>`;
    const action=kind==='people'?'Xem tiểu sử':kind==='videos'?'Xem video':kind==='media'?'Xem ảnh':'Đọc tư liệu';
    return `<article class="gdd-catalog-card">
      <button class="gdd-catalog-card-cover" type="button" data-open-kind="${kind}" data-id="${escape(row.id)}" aria-label="${escape(action+' '+title)}">${poster}</button>
      <div class="gdd-catalog-card-body"><span class="gdd-catalog-card-type">${escape(subtitle||current.title)} · ${escape(date(row.created_at))}</span>
        <h2>${escape(title)}</h2><p>${escape(description)}</p>
        <button class="gdd-catalog-card-action" type="button" data-open-kind="${kind}" data-id="${escape(row.id)}">${escape(action)} <span aria-hidden="true">↗</span></button>
      </div></article>`;
  }
  function renderPages(total){
    const element=byId('gdd-cat-pages');
    if(total<=1){element.replaceChildren();return;}
    const pages=new Set([1,total,...Array.from({length:5},(_,i)=>filters.page+i-2).filter(i=>i>=1&&i<=total)]);
    const ordered=[...pages].sort((a,b)=>a-b);
    let html=`<button type="button" data-p="${filters.page-1}" ${filters.page===1?'disabled':''} aria-label="Trang trước">‹ Trước</button>`;
    for(let i=0;i<ordered.length;i++){
      if(i>0&&ordered[i]-ordered[i-1]>1)html+='<span class="gdd-catalog-ellipsis" aria-hidden="true">…</span>';
      const page=ordered[i];
      html+=`<button type="button" data-p="${page}" ${page===filters.page?'aria-current="page"':''} aria-label="Trang ${page}">${page}</button>`;
    }
    html+=`<button type="button" data-p="${filters.page+1}" ${filters.page===total?'disabled':''} aria-label="Trang tiếp">Tiếp ›</button>`;
    element.innerHTML=html;
  }
  async function openFallbackDetail(kind,summary){
    const overlay=byId('detail-overlay'),content=byId('detail-content');
    if(!overlay||!content)return;
    const cfg=window.SITE_CONFIG||{};
    let row=summary;
    try{
      const params=new URLSearchParams({select:'*',id:'eq.'+summary.id,published:'eq.true',limit:'1'});
      const response=await fetch(String(cfg.supabaseUrl).replace(/\/$/,'')+'/rest/v1/'+kind+'?'+params,
        {headers:{apikey:cfg.supabasePublishableKey,Accept:'application/json'},credentials:'omit'});
      if(!response.ok)throw Error('HTTP '+response.status);
      const data=await response.json();
      if(!data.length)throw Error('Không tìm thấy tư liệu đã công bố');
      row=data[0];
    }catch(e){console.warn('[Dấu Xưa Cửa Đạo] Không tải được chi tiết đầy đủ:',e);}
    // Nếu chỉ có tiêu đề và tóm tắt, thông báo chưa tải được đầy đủ thay vì coi đó là toàn văn.
    const title=kind==='people'?row.full_name:row.title;
    const picture=safeURL(row.cover_url||row.image_url);
    const original=safeURL(kind==='videos'?row.video_url:row.file_url);
    const body=kind==='people'?row.biography:kind==='media'?row.caption:row.body;
    const source=row.sources||'Chưa bổ sung nguồn tư liệu.';
    // Gắn hash trước khi dựng nội dung để bộ album hiện hành có thể tìm đúng hồ sơ.
    history.replaceState(null,'','#'+PREFIX[kind]+'/'+encodeURIComponent(row.slug));
    content.innerHTML=`<div class="eyebrow">${escape(current.title)}</div><h2 class="detail-title">${escape(title)}</h2>
      <p class="detail-meta">${escape(row.rank||row.category||row.document_number||'')}</p>
      ${picture?`<div class="detail-cover"><img src="${escape(picture)}" alt="${escape(title)}"></div>`:''}
      ${row.summary?`<p class="detail-summary">${escape(row.summary)}</p>`:''}
      <div class="detail-body" style="white-space:pre-wrap">${escape(body||'Nội dung đầy đủ hiện chưa tải được. Vui lòng thử lại sau.')}</div>
      ${original?`<p class="detail-external"><a class="btn btn-primary" href="${escape(original)}" target="_blank" rel="noopener noreferrer">Mở tư liệu gốc ↗</a></p>`:''}
      <div class="detail-source"><strong>NGUỒN TƯ LIỆU &amp; GHI CHÚ</strong>${escape(source)}</div>`;
    overlay.hidden=false;document.body.classList.add('modal-open');
    overlay.querySelector('.detail-panel')?.scrollTo(0,0);
  }
  if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',buildUI,{once:true});
  else buildUI();
